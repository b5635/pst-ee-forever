for l in open("ee/dump/names.txt", encoding="utf-8", errors="replace"):
    ea, n, d = l.rstrip("\n").split("\t")

for line in body.splitlines():
    m = re.match(r"\s*case (0x[0-9A-F]+|\d+)u:", line)
    if m: cur.append(int(m.group(1), 0)); continue
    m = re.search(r"= off_([0-9A-F]+);", line) or re.search(r"= &`vtable for'(\w+)", line)
    if m and cur:
        try: name = vt.get(int(m.group(1), 16), "off_" + m.group(1))
        except ValueError: name = m.group(1)
        for c in cur: res.setdefault(c, name)
    if re.match(r"\s*break;", line) or re.match(r"\s*return", line): cur = []

# Opcode audit: every effect opcode PST data uses (SPL/ITM/EFF/CRE), mapped to GemRB's PST
# effects.ids name and the fx_ function GemRB binds it to; flags stub/partial implementations.

def effects_v1(d, off, n):
	for i in range(n):
		eo = off + i*0x30
		if eo + 0x30 > len(d): return
		op, tgt, pw, p1, p2, tm, rs, dur, pmax, pmin = struct.unpack_from("<HBBIIBBIBB", d, eo)
		res = d[eo+0x14:eo+0x1C].split(b"\0")[0].decode("latin1")
		yield dict(op=op, p1=p1, p2=p2, tm=tm, dur=dur, res=res)

uses = collections.defaultdict(list)   # op -> [(resource, p1, p2, res)]
for n in g.names("spl"):
	d = g.res(n, "spl")
	if not d or d[:4] != b"SPL ": continue
	eho, ehc, fxo, cfi, cfc = struct.unpack_from("<IHIHH", d, 0x64)
	for e in effects_v1(d, fxo + cfi*0x30, cfc): uses[e["op"]].append((n+".spl", e))
	for h in range(ehc):
		nfx, fi = struct.unpack_from("<HH", d, eho + h*0x28 + 0x1E)
		for e in effects_v1(d, fxo + fi*0x30, nfx): uses[e["op"]].append((n+".spl", e))
for n in g.names("itm"):
	d = g.res(n, "itm")
	if not d or d[:4] != b"ITM ": continue
	eho, ehc, fxo, fei, fec = struct.unpack_from("<IHIHH", d, 0x64)
	for e in effects_v1(d, fxo + fei*0x30, fec): uses[e["op"]].append((n+".itm", e))
	for h in range(ehc):
		nfx, fi = struct.unpack_from("<HH", d, eho + h*0x38 + 0x1E)
		for e in effects_v1(d, fxo + fi*0x30, nfx): uses[e["op"]].append((n+".itm", e))
for n in g.names("eff"):
	d = g.res(n, "eff")

		op, = struct.unpack_from("<I", d, 0x10)

		res = d[0x30:0x38].split(b"\0")[0].decode("latin1")
		uses[op].append((n+".eff", dict(op=op, p1=p1, p2=p2, res=res)))

# Align effects of the same SPL/ITM in original PST and PST:EE -> which EE opcode each PST opcode became
import pstres, struct, json, collections
o, e = pstres.orig(), pstres.ee()
ee_cls = {int(k): v for k, v in json.load(open("ee_opcodes.json")).items()}
def sig(d, eo):
    op, tg, pw, p1, p2, tm, rs, dur = struct.unpack_from("<HBBIIBBI", d, eo)
    res = d[eo+0x14:eo+0x1C].split(bytes(1))[0]
    return (op, (p1, p2, tm, dur, res))
import difflib
def effs(d, kind):
    if not d: return None
    if kind == "spl":
        if d[:4] != b"SPL ": return None
        eho, ehc, fxo, cfi, cfc = struct.unpack_from("<IHIHH", d, 0x64); hs = 0x28
    else:
        if d[:4] != b"ITM ": return None
        eho, ehc, fxo, cfi, cfc = struct.unpack_from("<IHIHH", d, 0x64); hs = 0x38
    groups = [[sig(d, fxo + (cfi + i) * 0x30) for i in range(cfc)]]
    for h in range(ehc):
        nfx, fi = struct.unpack_from("<HH", d, eho + h * hs + 0x1E)
        groups.append([sig(d, fxo + (fi + i) * 0x30) for i in range(nfx)])
    return groups
pairs = collections.defaultdict(collections.Counter); ex = {}
changed_files = []
for kind in ("spl", "itm"):
    for n in sorted(set(o.names(kind)) & set(e.names(kind))):
        a, b = effs(o.res(n, kind), kind), effs(e.res(n, kind), kind)
        if not a or not b: continue
        if [len(x) for x in a] != [len(x) for x in b]:
            changed_files.append((n + "." + kind, [len(x) for x in a], [len(x) for x in b]))
        for ga, gb in zip(a, b):
            sm = difflib.SequenceMatcher(None, [x[1] for x in ga], [y[1] for y in gb], autojunk=False)
            for bl in sm.get_matching_blocks():
                for k in range(bl.size):
                    x, y = ga[bl.a + k][0], gb[bl.b + k][0]
                    pairs[x][y] += 1; ex.setdefault((x, y), n + "." + kind)
out = {}
for x in sorted(pairs):
    out[x] = [(y, c, ee_cls.get(y, "?"), ex[(x, y)]) for y, c in pairs[x].most_common()]
json.dump({"map": {str(k): v for k, v in out.items()}, "changed": changed_files}, open("opmap.json", "w"), indent=1)
for x, v in out.items(): print(x, v[:3])
print("files with different effect counts:", len(changed_files))

import struct
class PE:
    def __init__(s, path):
        s.d = open(path, "rb").read(); d = s.d
        pe, = struct.unpack_from("<I", d, 0x3C); ns, = struct.unpack_from("<H", d, pe + 6); opt, = struct.unpack_from("<H", d, pe + 20)
        s.base, = struct.unpack_from("<I", d, pe + 24 + 28); s.secs = []
        so = pe + 24 + opt
        for i in range(ns):
            name, vs, va, rs, ra = struct.unpack_from("<8sIIII", d, so + i*40); s.secs.append((va, max(vs, rs), ra))
    def off(s, va):
        rva = va - s.base
        for v, sz, ra in s.secs:
            if v <= rva < v + sz: return rva - v + ra
    def bytes(s, va, n): o = s.off(va); return s.d[o:o+n]

def pro_refs(name, seen=None):
    seen = seen if seen is not None else set()

        refs |= {rr(d, 0x204), rr(d, 0x200 + 0x0C), rr(d, 0x200 + 0x16)}

        pass
    refs = {r for r in refs if r and re.match(r"^[A-Z0-9_#]+$", r)}

    out = set(refs)

# Minimal Infinity Engine resource reader for comparing PST data sets (original, EE, GemRB unhardcoded).
import os, struct, glob, zlib

EXT2T = {"bmp":1,"mve":2,"wav":4,"wfx":5,"plt":6,"bam":1000,"wed":1001,"chu":1002,"tis":1003,"mos":1004,
	"itm":1005,"spl":1006,"bcs":1007,"ids":1008,"cre":1009,"are":1010,"dlg":1011,"2da":1012,"gam":1013,
	"sto":1014,"wmp":1015,"eff":1016,"bs":1017,"chr":1018,"vvc":1019,"vef":1020,"pro":1021,"bio":1022,
	"wbm":1023,"fnt":1024,"gui":1026,"sql":1027,"pvrz":1028,"glsl":1029,"menu":1032,"lua":1033,"ttf":1034,"png":1035,
	"ini":2050,"src":1035}
T2EXT = {v:k for k,v in EXT2T.items()}

class Game:
	def __init__(self, path, tlk=None, override=True, extra=()):
		self.path = path
		key = None
		for n in ("chitin.key","CHITIN.KEY"):
			if os.path.exists(os.path.join(path,n)): key = os.path.join(path,n)
		k = open(key,"rb").read()
		nbif, nres, obif, ores = struct.unpack_from("<IIII", k, 8)
		self.bifs = []
		for i in range(nbif):
			ln, fo, fl, loc = struct.unpack_from("<IIHH", k, obif + i*12)
			self.bifs.append(k[fo:fo+fl].rstrip(b"\0").decode().replace("\\", "/"))
		self.index = {}
		for i in range(nres):
			rn, rt, loc = struct.unpack_from("<8sHI", k, ores + i*14)
			self.index[(rn.rstrip(b"\0").decode("latin1").upper(), rt)] = loc
		self.dirs = list(extra) + ([os.path.join(path,"override")] if override else [])
		self.dirfiles = {}
		for d in self.dirs:
			m = {}
			if os.path.isdir(d):
				for f in os.listdir(d):
					m[f.upper()] = os.path.join(d,f)
			self.dirfiles[d] = m
		self.bifcache = {}
		tlk = tlk or os.path.join(path,"dialog.tlk")
		self.tlk = open(tlk,"rb").read() if os.path.exists(tlk) else None

	def _bif(self, i):
		if i in self.bifcache: return self.bifcache[i]
		bp = self.bifs[i]
		cands = [os.path.join(self.path, bp)] + glob.glob(os.path.join(self.path, "**", os.path.basename(bp)), recursive=True)
		data = None
		for c in cands:
			if os.path.exists(c):
				data = open(c,"rb").read(); break
		if data and data[:8] == b"BIFCV1.0":
			ulen, = struct.unpack_from("<I", data, 8+4+len(data)) if False else (0,)
			nlen, = struct.unpack_from("<I", data, 8)
			o = 12 + nlen
			ulen, clen = struct.unpack_from("<II", data, o)
			data = zlib.decompress(data[o+8:o+8+clen])
		elif data and data[:8] == b"BIF V1.0":
			o = 12; out = b""
			while o < len(data):
				ul, cl = struct.unpack_from("<II", data, o); out += zlib.decompress(data[o+8:o+8+cl]); o += 8+cl
			data = out
		self.bifcache[i] = data
		return data

	def names(self, ext, with_override=True):
		t = EXT2T[ext]
		s = {n for (n,rt) in self.index if rt == t}
		if with_override:
			for d,m in self.dirfiles.items():
				for f in m:
					if f.endswith("." + ext.upper()): s.add(f[:-len(ext)-1])
		return sorted(s)

	def where(self, name, ext):
		fn = (name + "." + ext).upper()
		for d in self.dirs:
			if fn in self.dirfiles[d]: return d
		if (name.upper(), EXT2T[ext]) in self.index: return "bif"
		return None

	def res(self, name, ext, bif_only=False):
		fn = (name + "." + ext).upper()
		if not bif_only:
			for d in self.dirs:
				p = self.dirfiles[d].get(fn)
				if p: return open(p,"rb").read()
		loc = self.index.get((name.upper(), EXT2T[ext]))
		if loc is None: return None
		b = self._bif(loc >> 20)
		if not b: return None
		nf, nt, off = struct.unpack_from("<III", b, 8)
		if ext in ("tis",):
			toff = off + nf*16
			for j in range(nt):
				l,o,c,s,t = struct.unpack_from("<IIIIH", b, toff + j*20)
				if (l >> 14) == ((loc >> 14) & 0x3F): return b[o:o+c*s]
			return None
		for j in range(nf):
			l,o,s,t = struct.unpack_from("<IIIH", b, off + j*16)
			if (l & 0x3FFF) == (loc & 0x3FFF): return b[o:o+s]
		return None

	def string(self, ref):
		if not self.tlk or ref < 0 or ref == 0xFFFFFFFF: return ""
		cnt, toff = struct.unpack_from("<II", self.tlk, 0x0A)
		if ref >= cnt: return ""
		fl, snd, v, p, o, l = struct.unpack_from("<H8sIIII", self.tlk, 0x12 + ref*26)
		return self.tlk[toff+o:toff+o+l].decode("cp1252","replace")

def text(data):
	"""2DA/IDS text, decrypting 0xFFFF-prefixed files"""
	if data is None: return None
	if data[:2] == b"\xff\xff":
		key = b"88a88fa974e8ada31b6d5a2c8d58a96e5a2c8d58d6a5f4e5bd9e7b2adb3c5da0"
		key = bytes.fromhex("88A88FBA8AD3B9F5EDB1CFEAAAE4B5FBEB82F990CAC9B5E7DCAAE4B5FB8E8BF2")
		data = bytes(c ^ key[i % len(key)] for i,c in enumerate(data[2:]))
	return data.decode("latin1")
